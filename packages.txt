# Packages these dotfiles expect, one per line: name, then "# what it is for".
# `./bootstrap` reads this file; `./bootstrap --check` reports what is missing
# without installing anything. Names are Arch/CachyOS package names.
#
# [core]     the desktop does not work without these
# [fonts]    referenced by name in the shell (quickshell) and kitty
# [theming]  matugen recolours these apps; skip any you do not use
# [optional] tools bound to a key or a bar module; everything else keeps working
# [aur]      not in the official repos: needs paru/yay, or install by hand

[core]
hyprland                     # compositor; must be >= 0.56 (reads hyprland.lua natively)
hypridle                     # idle -> lock -> screen off
hyprlock                     # lock screen fallback when the shell is not running (SUPER+L, hypridle: qs ipc call lock lock || hyprlock)
quickshell                   # the shell: bar, popups, settings, notifications, launcher, clipboard, cheatsheet, wallpaper strip, OSD, lock screen, greeter
hyprpolkitagent              # polkit prompts; restarted by hyprland.lua at login
xdg-desktop-portal           # portals: file pickers, screenshare
xdg-desktop-portal-hyprland  # Hyprland portal backend (screenshare)
xdg-desktop-portal-gtk       # GTK portal backend (file dialogs, settings)
kitty                        # terminal (SUPER+Q), also hosts btop in the dashboard
wl-clipboard                 # wl-copy / wl-paste
cliphist                     # clipboard history behind the shell's clipboard picker (SUPER+SHIFT+C)
grim                         # screenshots
slurp                        # region selection for screenshots
pipewire                     # audio
pipewire-pulse               # PulseAudio shim for the apps
wireplumber                  # wpctl volume keys
playerctl                    # media keys
networkmanager               # the shell's network panel talks to it over D-Bus; nmcli for the hotspot
matugen                      # generates every colours file from the wallpaper
awww                         # wallpaper daemon (swww fork); setwall works without it, the picture just won't change
polkit                       # required by hyprpolkitagent
libnotify                    # notify-send, used by notify-unit-failure
fish                         # shell the config is written for
jq                           # relayout, gen-ohmyposh-template
python                       # matugenfox-host, reload-steam-css, matugen-icons
brightnessctl                # brightness keys
adw-gtk-theme                # GTK theme matugen's gtk3 hook selects (adw-gtk3-dark)
glib2                        # gsettings, used by that hook
adwaita-icon-theme           # source SVGs for the Matugen icon theme (matugen-icons recolours its folders)
breeze-icons                 # app icons the Matugen theme inherits
breeze                       # Qt widget style named in kdeglobals
qt6ct                        # QT_QPA_PLATFORMTHEME=qt6ct: Breeze style + the matugen palette for Qt apps (qt6ct/qt6ct.conf)
qqc2-breeze-style            # Breeze for QtQuick apps (kdeconnect etc.); qqc2-desktop-style comes with them
nemo                         # file manager on SUPER+E; GTK3, so it recolours live with the wallpaper
swayimg                      # image viewer: floats centred, colours from matugen (swayimg/init.lua), Enter = gallery
mpv                          # video player; the ModernZ OSC (aur) draws the Jellyfin-style bar
nemo-fileroller              # compress/extract in Nemo's context menu
nemo-terminal                # embedded terminal pane in Nemo (F4)
file-roller                  # archive manager nemo-fileroller drives
gvfs-mtp                     # phones in Nemo's sidebar
gvfs-smb                     # network shares in Nemo's sidebar

[fonts]
otf-geist-mono-nerd          # primary UI font: the shell
ttf-meslo-nerd               # fallback Nerd Font for the same
ttf-jetbrains-mono-nerd      # kitty font
noto-fonts                   # GTK default UI font (settings.ini)

[theming]
btop                         # matugen theme + live reload on SIGUSR2 (btop >= 1.4.7)
spotify-launcher             # Spotify; theme applied through spicetify (see README)
steam                        # skin via Adwaita-for-Steam (bootstrap clones it) + live push
firefox                      # site colours via the MatugenFox extension (see README)
capitaine-cursors            # cursor theme named in gtk settings.ini

[optional]
fastfetch                    # Settings > About PC reads its JSON (board, BIOS, GPU drivers, disks, displays); shorter list without it
webp-pixbuf-loader           # only if your wallpapers are .webp
nwg-displays                 # GUI to regenerate monitors.lua for your monitors
wayvnc                       # remote desktop over Tailscale (remote-desktop on; the network panel's Remote desktop toggle)
openssh                      # ssh into the desktop over Tailscale only (./sshd install: AllowUsers on the tailnet ranges, ufw rule on tailscale0)
python-websockets            # reload-steam-css (Steam DevTools push)
profile-sync-daemon          # not part of the dotfiles; on the reference machine only
greetd                       # login manager; runs the shell's greeter (quickshell from [core]). ./greeter installs and enables it
xsettingsd                   # GTK theme/icons/cursor for Xwayland apps (Steam, JetBrains); xsettingsd.service is hooked into hyprland-session.target by bootstrap

[aur]
mpv-modernz-git              # mpv OSC: title, rounded seekbar, centred play/skip, sub/audio buttons; colours via matugen (templates/modernz.conf)
mpv-thumbfast-git            # seekbar hover thumbnails for ModernZ
slack-desktop                # dashboard app in relayout; dropped from the layout when absent
google-chrome                # chrome-flags.conf; harmless without it (links follow mimeapps.list, firefox)
oh-my-posh-bin               # prompt; fish config skips it when missing (see README for the theme file)
vesktop                      # Discord client; quickCss.css is generated (needs useQuickCss on). In CachyOS's own repo, AUR everywhere else
vscodium-bin                 # editor theme via the haikalllp.matugen-theme extension (watches ~/.cache/matugen). CachyOS ships `vscodium`; either is fine
